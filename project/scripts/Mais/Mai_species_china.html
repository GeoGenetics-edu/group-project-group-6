wilcox_results <- dat_rel %>%
  group_by(taxon, Genus, Species) %>%
  summarise(
    p_value = wilcox.test(
      rel_abund ~ Disease.MESH.ID,
      exact = FALSE
    )$p.value,

    mean_D003924 = mean(
      rel_abund[Disease.MESH.ID == "D003924"]
    ),

    mean_D006262 = mean(
      rel_abund[Disease.MESH.ID == "D006262"]
    ),

    log2_fc = log2(
      (mean_D003924 + 1e-6) /
      (mean_D006262 + 1e-6)
    ),

    .groups = "drop"
  ) %>%
  mutate(
    q_value = p.adjust(p_value, method = "BH")
  )
wilcox_D006262_enriched <- wilcox_results %>%
